139inline std::string dump(const common::AppEvent& ev) {
140 return std::visit(
141 Overloaded{
142 [](const ButtonPressedEvent& e) {
143 return "ButtonPressedEvent{b=" + dump(e.button) + "}";
144 },
145 [](const ButtonLongPressedEvent& e) {
146 return "ButtonLongPressedEvent{b=" + dump(e.button) + "}";
147 },
148 [](const PlaybackStatusChangedEvent& e) {
149 return "PlaybackStatusChangedEvent{s=" + dump(e.status) + "}";
150 },
151 [](const TempHumidUpdateEvent& e) {
152 return "TempHumidUpdateEvent{t=" + std::to_string(e.temperature) +
153 ", h=" + std::to_string(e.humidity) + "}";
154 },
155 [](const LightLevelUpdateEvent& e) {
156 return std::string("LightLevelUpdateEvent{lux=") + std::to_string(e.lux) + "}";
157 },
158 [](const BatteryLevelUpdateEvent& e) {
159 return std::string("BatteryLevelUpdateEvent{mv=") + std::to_string(e.millivolts) +
160 ", p=" + std::to_string(e.percent) + "}";
161 },
162 [](const SystemInitedEvent&) { return std::string("SystemInitedEvent{}"); },
163 [](const ClapFeatureStateChangedEvent& c) {
164 return std::string("ClapFeatureStateChangedEvent{e=") +
165 (c.isEnabled ? "true}" : "false}");
166 },
167 [](const CurrentStationChangedEvent&) {
168 return std::string{"CurrentStationChangedEvent{}"};
169 },
170 [](const WifiStateChangedEvent& e) {
171 return std::string("WifiStateChangedEvent{c=") +
172 (e.isConnected ? "true, b=" : "false, b=") + std::to_string(e.bars) + "}";
173 },
174 [](const VolumeChangedEvent& e) {
175 return std::string("VolumeChangedEvent{v=") + std::to_string(e.volume) + "}";
176 },
177 [](const SwitchToWifiProvScreenEvent&) {
178 return std::string("SwitchToWifiProvScreenEvent{}");
179 },
180 [](const SwitchToSyncInProgressScreenEvent&) {
181 return std::string("SwitchToSyncInProgressScreenEvent{}");
182 },
183 [](const SwitchToMainScreenEvent&) { return std::string("SwitchToMainScreenEvent{}"); },
184 [](const WifiCredsReceivedEvent& e) { return std::string("WifiCredsReceivedEvent{}"); },
185 [](const auto&) { return std::string{"<unknown event>"}; }},
186 ev);
187}
